    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    function playSound(type) {
        if (audioCtx.state === 'suspended') audioCtx.resume();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        if (type === 'coin') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(987.77, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(1318.51, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.15);
        } else if (type === 'correct') {
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
            osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.1);
            osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.2);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.35);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.35);
        } else if (type === 'wrong') {
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(220, audioCtx.currentTime);
            osc.frequency.setValueAtTime(180, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.3);
        }
    }

    let gameCanvas, gameCtx;
    let basket = { x: 180, y: 320, width: 70, height: 40 };
    let fallingItems = [];
    let gameScore = 0;
    let gameInterval = null;
    let itemSpawnInterval = null;

    function initMinigame() {
        gameCanvas = document.getElementById('gameCanvas');
        gameCtx = gameCanvas.getContext('2d');
        gameCanvas.width = 400;
        gameCanvas.height = 380;

        basket.x = gameCanvas.width / 2 - basket.width / 2;
        fallingItems = [];
        gameScore = 0;
        document.getElementById('game-score').innerText = '0';
        document.getElementById('game-over-modal').classList.add('hidden');

        if (gameInterval) clearInterval(gameInterval);
        if (itemSpawnInterval) clearInterval(itemSpawnInterval);

        gameInterval = setInterval(updateMinigame, 1000 / 60);
        itemSpawnInterval = setInterval(spawnFallingItem, 900);

        setTimeout(endMinigame, 30000);
    }

    function spawnFallingItem() {
        const isCora = Math.random() > 0.3;
        fallingItems.push({
            x: Math.random() * (gameCanvas.width - 30),
            y: -30,
            type: isCora ? 'cora' : 'rock',
            speed: 2 + Math.random() * 3
        });
    }

    function updateMinigame() {
        gameCtx.clearRect(0, 0, gameCanvas.width, gameCanvas.height);

        gameCtx.fillStyle = '#FFD166';
        gameCtx.beginPath();
        gameCtx.roundRect(basket.x, basket.y, basket.width, basket.height, 12);
        gameCtx.fill();
        gameCtx.strokeStyle = '#B45309';
        gameCtx.lineWidth = 3;
        gameCtx.stroke();
        
        gameCtx.fillStyle = '#78350F';
        gameCtx.font = 'bold 12px Fredoka';
        gameCtx.fillText('🧺 Ahorro', basket.x + 12, basket.y + 24);

        for (let i = fallingItems.length - 1; i >= 0; i--) {
            let item = fallingItems[i];
            item.y += item.speed;

            if (item.type === 'cora') {
                gameCtx.fillStyle = '#38BDF8';
                gameCtx.beginPath();
                gameCtx.arc(item.x + 15, item.y + 15, 14, 0, Math.PI * 2);
                gameCtx.fill();
                gameCtx.fillStyle = '#0369A1';
                gameCtx.font = 'bold 10px Fredoka';
                gameCtx.fillText('25¢', item.x + 7, item.y + 18);
            } else {
                gameCtx.fillStyle = '#64748B';
                gameCtx.beginPath();
                gameCtx.arc(item.x + 15, item.y + 15, 12, 0, Math.PI * 2);
                gameCtx.fill();
                gameCtx.fillStyle = '#FFFFFF';
                gameCtx.font = 'bold 12px sans-serif';
                gameCtx.fillText('🪨', item.x + 7, item.y + 20);
            }

            if (
                item.y + 30 >= basket.y &&
                item.x + 30 >= basket.x &&
                item.x <= basket.x + basket.width
            ) {
                if (item.type === 'cora') {
                    playSound('coin');
                    gameScore += 1;
                    document.getElementById('game-score').innerText = gameScore;
                } else {
                    playSound('wrong');
                    gameScore = Math.max(0, gameScore - 1);
                    document.getElementById('game-score').innerText = gameScore;
                }
                fallingItems.splice(i, 1);
            } else if (item.y > gameCanvas.height) {
                fallingItems.splice(i, 1);
            }
        }
    }

    function moveBasketLeft() {
        basket.x = Math.max(0, basket.x - 35);
    }

    function moveBasketRight() {
        basket.x = Math.min(gameCanvas.width - basket.width, basket.x + 35);
    }

    }

    async function handleTutorQuery(e) {
        e.preventDefault();
        const inputEl = document.getElementById('tutor-input');
        const query = inputEl.value.trim();
        if (!query) return;

        const chatBox = document.getElementById('chat-messages');

        chatBox.innerHTML += `
            <div class="flex items-start justify-end space-x-2">
                <div class="bg-brand-blue text-white p-3 rounded-2xl text-xs sm:text-sm shadow-sm max-w-[85%] font-semibold">
                    ${query}
                </div>
            </div>
        `;
        inputEl.value = '';
        chatBox.scrollTop = chatBox.scrollHeight;

        const loadingId = 'loading-' + Date.now();
        chatBox.innerHTML += `
            <div id="${loadingId}" class="flex items-start space-x-2">
                <div class="text-2xl">🐱</div>
                <div class="bg-amber-100 text-amber-900 p-3 rounded-2xl text-xs font-bold animate-pulse">
                    Cat-Tal está pensando su respuesta salvadoreña... 🇸🇻
                </div>
            </div>
        `;
        chatBox.scrollTop = chatBox.scrollHeight;

        try {
            const systemPrompt = `Eres Cat-Tal, un simpático gato tutor financiero para niños de San Salvador, El Salvador. Responde en español salvadoreño amigable, alegre y educativo (máximo 3 oraciones). Explica el dinero en USD (coras, centavos, billetes) usando ejemplos como pupusas, buses o tiendas. El usuario se llama ${userProfile.username} (${userProfile.age} años).`;
            
            const apiKey = ""; 
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: query }] }],
                    systemInstruction: { parts: [{ text: systemPrompt }] }
                })
            });

            const data = await response.json();
            document.getElementById(loadingId).remove();

            let replyText = "¡Recuerda que 4 coras de 25 centavos hacen un Dólar ($1.00)! Sigue ahorrando.";
            if (data.candidates && data.candidates[0]?.content?.parts[0]?.text) {
                replyText = data.candidates[0].content.parts[0].text;
            }

            chatBox.innerHTML += `
                <div class="flex items-start space-x-2">
                    <div class="text-2xl">🐱</div>
                    <div class="bg-white p-3 rounded-2xl border border-slate-200 text-xs sm:text-sm shadow-sm max-w-[85%] text-slate-800">
                        ${replyText}
                    </div>
                </div>
            `;
            chatBox.scrollTop = chatBox.scrollHeight;
            speakText(replyText);

        } catch (err) {
            console.error(err);
            document.getElementById(loadingId).remove();
            chatBox.innerHTML += `
                <div class="flex items-start space-x-2">
                    <div class="text-2xl">🐱</div>
                    <div class="bg-white p-3 rounded-2xl border border-slate-200 text-xs sm:text-sm shadow-sm max-w-[85%] text-slate-800">
                        Una cora equivale a 25 centavos de dólar ($0.25). ¡Guárdala en tu alcancía! 🪙
                    </div>
                </div>
            `;
        }
    }
